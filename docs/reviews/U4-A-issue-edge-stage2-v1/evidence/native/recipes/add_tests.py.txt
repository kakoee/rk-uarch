fn stage2_result(raw: &str) -> Value {
    let v: Value = serde_json::from_str(raw).unwrap();
    execute_with_observations(raw, &anchor(&v), ObservationVersion::V3).unwrap()
}
fn stage2_mixed_causes(raw: &str, release: u64) {
    let r=stage2_result(raw);
    let rows=r["observations"]["activities"]["records"].as_array().unwrap();
    let opportunities=r["observations"]["reservation_opportunities"]["records"].as_array().unwrap();
    for (request,cursor,expected) in [("r02.00",52,vec![("r00","read_port",50)]),("r03.00",56,vec![("r00","read_port",50),("r01","ack",55)])] {
        let opportunity=opportunities.iter().find(|o|o["granted_request_ids"].as_array().unwrap().iter().any(|x|x==request)).unwrap();
        assert_eq!(opportunity["event_cursor"],cursor);
        assert_eq!(opportunity["t_ps"],release);
        let alignment=rows.iter().find(|a|a["request_id"]==request && a["subphase"]=="dma_issue" && a["kind"]=="alignment").unwrap();
        let mut causes=alignment["predecessors"].as_array().unwrap().iter().filter(|p|p["relation"]=="credit").map(|p|{
            let a=rows.iter().find(|a|a["activity_id"]==p["activity_id"]).unwrap();
            assert_eq!(a["kind"],"service");
            assert_eq!(a["end_ps"],release);
            assert!(a["completion_ordinal"].as_u64().unwrap()<cursor);
            (a["job_id"].as_str().unwrap(),a["subphase"].as_str().unwrap(),a["completion_ordinal"].as_u64().unwrap())
        }).collect::<Vec<_>>();causes.sort();assert_eq!(causes,expected);
    }
}
