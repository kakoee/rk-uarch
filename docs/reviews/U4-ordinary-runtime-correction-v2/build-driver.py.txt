"""Task-owned offline native commands; cleanup only the full ID returned by create."""
import json, os, re, subprocess, sys
from pathlib import Path
BASE=Path('/tmp/rk-uarch-u4-correction-build')
BASE.mkdir(exist_ok=True)
(BASE/'target').mkdir(exist_ok=True)
IMAGE='sha256:e9f69f41498334be985ed4d25973fe3fa3f6519e58c73e6a7b8397d871fa0523'
BIN='/usr/local/rustup/toolchains/1.99.0-x86_64-unknown-linux-gnu/bin/'
mode=sys.argv[1]
commands={
 'fmt':['cargo','fmt','--all'],
 'check':['cargo','check','--locked','--offline','--target','x86_64-unknown-linux-gnu'],
 'debug':['cargo','build','--locked','--offline','--target','x86_64-unknown-linux-gnu'],
 'release':['cargo','build','--release','--locked','--offline','--target','x86_64-unknown-linux-gnu'],
 'test':['cargo','test','--locked','--offline','--all-features','--target','x86_64-unknown-linux-gnu'],
 'clippy':['cargo','clippy','--locked','--offline','--all-targets','--target','x86_64-unknown-linux-gnu','--','-D','warnings'],
 'fmt-check':['cargo','fmt','--all','--check'],
 'versions':['rustc','--version'],
}
args=['docker','create','--pull','never','--network','none','--read-only','--user',f'{os.getuid()}:{os.getgid()}',
      '--cap-drop','ALL','--security-opt','no-new-privileges','--tmpfs','/tmp','--workdir','/work/native',
      '--mount','type=bind,src=/tmp/rk-uarch-u4-correction,dst=/work'+('' if mode=='fmt' else ',readonly'),
      '--mount','type=bind,src=/tmp/u4-k2-c1-precise.styffox_/cargo-home,dst=/cargo-home,readonly',
      '--mount',f'type=bind,src={BASE}/target,dst=/target',
      '--entrypoint','/usr/bin/env',IMAGE,'-i','PATH='+BIN+':/usr/local/cargo/bin:/usr/bin:/bin',
      'CARGO_HOME=/cargo-home','CARGO_TARGET_DIR=/target','CARGO_NET_OFFLINE=true','RUSTUP_AUTO_INSTALL=0',
      'RUSTC='+BIN+'rustc','LC_ALL=C']+commands[mode]
cid=subprocess.check_output(args,text=True).strip()
assert re.fullmatch('[0-9a-f]{64}',cid), cid
with (BASE/'containers.jsonl').open('a') as f: f.write(json.dumps({'id':cid,'command':commands[mode]})+'\n')
try:
    with (BASE/(mode+'.log')).open('w') as log:
        result=subprocess.run(['docker','start','--attach',cid],stdout=log,stderr=subprocess.STDOUT,timeout=900)
    state=json.loads(subprocess.check_output(['docker','inspect',cid]))[0]['State']
    record={'command':commands[mode],'attach_status':result.returncode,'state':state,'container_id':cid}
    (BASE/(mode+'.json')).write_text(json.dumps(record,indent=2)+'\n')
    print((BASE/(mode+'.log')).read_text()[-14000:])
    if result.returncode or state['Running'] or state['ExitCode'] or state['OOMKilled'] or state['Error']:
        raise SystemExit(1)
finally:
    subprocess.run(['docker','rm','--force',cid],check=True,stdout=subprocess.DEVNULL)
