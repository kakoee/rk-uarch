PS = 10**12


def edge(freq_hz: int, k: int) -> int:
    """Build-spec section 2.8: t_k = ceil(k * 10^12 / freq_hz)."""
    return -(-k * PS // freq_hz)


def first_edge_index(freq_hz: int, t_ps: int) -> int:
    """Definition by search: smallest k with edge(k) >= t."""
    k = (t_ps * freq_hz) // PS
    while k > 0 and edge(freq_hz, k - 1) >= t_ps:
        k -= 1
    while edge(freq_hz, k) < t_ps:
        k += 1
    return k


def next_edge(freq_hz: int, t_ps: int, n: int) -> int:
    return edge(freq_hz, first_edge_index(freq_hz, t_ps) + n)


def proposal_index(freq_hz: int, t_ps: int) -> int:
    """verification.md E2 integer formula."""
    return 0 if t_ps == 0 else (t_ps - 1) * freq_hz // PS + 1

def ws_cycles(m: int, k: int, n: int) -> tuple[int, int, int, int]:
    x, w, y = 2 * m * k, 2 * k * n, 2 * m * n
    stage = -(-(x + w) // 1024)
    wave = 255 + m + 255
    out = -(-y // 1024)
    return 32 + stage + wave + out, stage, wave, out

def read_2hop(n_same_channel: int, f_core: int, f_noc: int, f_dram: int, reading: str) -> list[int]:
    """Completion times of n reads issued at t=0 to one 2-hop channel (4 NoC cycles each way).

    reading="doc": hop delay added as fixed ps (4 * period), channel finish ceil-to-ps.
    reading="next_edge": every cycle count converted via next_edge in its own domain,
                         channel finish ceil-to-ps (bandwidth service is not in cycles).
    reading="next_edge+dram_edge": as next_edge, and channel completion aligned up to a DRAM edge.
    """
    out = []
    for i in range(1, n_same_channel + 1):
        if reading == "doc":
            arrive = 4 * PS // f_noc  # exact here only because PS/f_noc is integral in the cases used
        else:
            arrive = next_edge(f_noc, 0, 4)
        lat_end = next_edge(f_dram, arrive, 48)
        chan = math.ceil(lat_end + i * svc)
        if reading == "next_edge+dram_edge":
            chan = next_edge(f_dram, chan, 0)
        if reading == "doc":
            resp = chan + 4 * PS // f_noc
        else:
            resp = next_edge(f_noc, chan, 4)
        core_edge = next_edge(f_core, resp, 0)
        out.append((chan, resp, next_edge(f_core, core_edge, 1)))
    return out

    return next_edge(GHZ, next_edge(GHZ, resp, 0), 1)

# ---------------------------------------------------------------- zero-delay convention
def zero_delay(reading: str) -> int:
    """owner0: a=3000 ready 0. owner1: p=1000 ready 0. z: 0 ps on owner0, after p.
    c: 2000 ps on owner1, after z.  'inline' completes z at readiness; 'owner' waits for owner0."""
    p_end = 1000
    z_end = p_end if reading == "inline" else max(p_end, 3000)
    return max(3000, z_end + 2000)


print("ZERO-DELAY-OWNED", {r: zero_delay(r) for r in ("inline", "owner")}, "V26-LITERAL", 0 + 0 + 3000)

def rope(M, Hq, Hkv, D, R, w, casts):
    vec = (3 + (2 if casts else 0)) * M * (Hq + Hkv) * R
    rd = w * M * (Hq + Hkv) * D + 8 * M * R
    wr = w * M * (Hq + Hkv) * D
    return vec, rd, wr

def rot(x, cos, sin, pairs):
    y = list(x)
    for j, (i0, i1) in enumerate(pairs):
        a_, b_ = x[i0], x[i1]
        y[i0], y[i1] = a_ * cos[j] - b_ * sin[j], a_ * sin[j] + b_ * cos[j]
    return y

      rot([1, 2, 3, 4], [0, 1], [1, 0], [(0, 2), (1, 3)]))

# Smallest same-channel burst (2-hop, t=0 issue, ample credits) whose final DMA completion
# times differ between the derated law and the omitted-refresh mutant. Port service is one
# core cycle per response; responses here are >= 2 ns apart, so the port never queues.
