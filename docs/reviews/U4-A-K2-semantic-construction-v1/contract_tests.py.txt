import pytest
from typing import Literal, cast

from uarch_contract.mapped import ClockObservation, CriticalInterval, EventObservation, QueueObservation


def test_public_raw_edge_and_owned_zero_are_distinct_fields() -> None:
    value = dict(job_id="z", domain="core", grant_ps=835, n_cycles=0,
                 raw_edge_outcome="success", aligned_start_ps=1667, completion_ps=835)
    assert ClockObservation.model_validate(value).completion_ps == 835
    with pytest.raises(ValueError):
        ClockObservation.model_validate(dict(value, n_cycles=True))
    with pytest.raises(ValueError):
        ClockObservation.model_validate(dict(value, raw_edge_outcome="TimeOverflow"))


def test_diagnostic_overflow_is_explicit_not_zero_job_failure() -> None:
    maximum = 2**64 - 1
    value = ClockObservation(job_id="z", domain="core", grant_ps=maximum, n_cycles=0,
        raw_edge_outcome="TimeOverflow", aligned_start_ps=None, completion_ps=maximum)
    assert value.aligned_start_ps is None


@pytest.mark.parametrize("kind", ["TASK_RELEASE", "DEPENDENCY_READY", "OWNER_GRANT"])
def test_complete_public_event_vocabulary(
    kind: Literal["TASK_RELEASE", "DEPENDENCY_READY", "OWNER_GRANT"]
) -> None:
    value = EventObservation(event_seq=0, observation_ordinal=0, t_ps=835,
        phase=2 if kind == "OWNER_GRANT" else 1, target_id=0, kind=kind, job_id="z")
    assert value.kind == kind


def test_idle_is_distinct_from_resource_service() -> None:
    idle = dict(kind="release_idle", start_ps=0, end_ps=835, cause_job_id="z",
                predecessor_id=None, limiting_resource_id=None, category="overhead")
    assert CriticalInterval.model_validate(idle).limiting_resource_id is None
    for update in ({"limiting_resource_id": 0}, {"predecessor_id": "p"},
                   {"category": "compute"}, {"start_ps": 1}, {"end_ps": 0}):
        with pytest.raises(ValueError):
            CriticalInterval.model_validate(dict(idle, **update))
    with pytest.raises(ValueError):
        CriticalInterval.model_validate(dict(idle, kind="service"))
    assert CriticalInterval.model_validate(dict(idle, kind="service", limiting_resource_id=0))


def test_non_clocked_fifo_does_not_invent_domain() -> None:
    value = QueueObservation(queue_id="owner:0:waiting", item_id="z", population="waiting",
        enter_ps=835, leave_ps=2000, owner_id=0, clock_domain=None)
    assert value.clock_domain is None

@pytest.mark.parametrize("release,aligned", [(0, 0), (835, 1667)])
def test_synthetic_accounting_subrecords_and_faults(release: int, aligned: int) -> None:
    # Deliberately partial namespaces: cannot serialize as MappedEngineResult/native evidence.
    import copy
    from typing import Any
    from types import SimpleNamespace as NS
    from uarch_contract.mapped import (
        ClockSetup, FixedDelay, JobObservation, Observed,
        validate_fixed_delay_observations,
    )
    setup = ClockSetup(job_id="z", domain="core", start_ps=release, n_cycles=0)
    declared = FixedDelay(id="z", duration_ps=0, owner_id=0, kind="fixed_delay")
    job: Any = NS(clock_setup=(setup,), freq_hz=NS(core=1200000000),
             mapped_point=NS(task_graph=NS(jobs=(declared,), dependencies=())))
    record = JobObservation(job_id="z", op_id=None, group_id=None, repeat_index=None,
        tile_origin=None, tile_extents=None, release_ps=release, ready_ps=release,
        start_ps=release, end_ps=release, owner_id=0, blocking_predecessor_ids=(),
        useful_matrix_ops=0, issued_matrix_ops=0, vector_arithmetic_ops=0, cast_ops=0)
    kinds: tuple[Literal["TASK_RELEASE", "TASK_READY", "OWNER_GRANT", "COMPUTE_DONE"], ...] = (
        "TASK_RELEASE" if release else "TASK_READY", "OWNER_GRANT", "COMPUTE_DONE"
    )
    events = tuple(EventObservation(event_seq=i, observation_ordinal=i, t_ps=release,
        phase=1 if i == 0 else 2, target_id=0, kind=kind, job_id="z")
        for i, kind in enumerate(kinds))
    clock = ClockObservation(job_id="z", domain="core", grant_ps=release, n_cycles=0,
        raw_edge_outcome="success", aligned_start_ps=aligned, completion_ps=release)
    queues = tuple(QueueObservation(queue_id="owner:0:" + p, item_id="z", population=p,
        enter_ps=release, leave_ps=release, owner_id=0, clock_domain=None)
        for p in ("service", "waiting"))
    critical = (CriticalInterval(kind="release_idle", start_ps=0, end_ps=release,
        cause_job_id="z", predecessor_id=None, limiting_resource_id=None, category="overhead"),
        ) if release else ()
    r: Any = NS(duration_ps=release, simulator_metrics=NS(event_count=3, max_live_events=1),
        observations=NS(dma=NS(status="unrepresented"), transfers=NS(status="unrepresented"),
            allocations=NS(status="unrepresented"), jobs=NS(records=(record,)), events=NS(records=events),
            clocks=Observed[ClockObservation](status="observed", records=(clock,)),
            queue_membership=Observed[QueueObservation](status="observed", records=queues),
            resource_intervals=NS(records=()), critical_path=NS(records=critical)))
    validate_fixed_delay_observations(r, job)
    for fault in ("alignment", "domain", "blocker", "critical", "dispatch", "metrics"):
        bad = copy.deepcopy(r)
        if fault == "alignment":
            bad.observations.clocks = Observed[ClockObservation](status="observed",
                records=(clock.model_copy(update={"aligned_start_ps": aligned + 1}),))
        elif fault == "domain":
            bad.observations.queue_membership = Observed[QueueObservation](status="observed",
                records=(queues[0].model_copy(update={"clock_domain": "core"}), queues[1]))
        elif fault == "blocker":
            bad.observations.jobs.records = (record.model_copy(update={"blocking_predecessor_ids": ("z",)}),)
        elif fault == "critical":
            bad.observations.critical_path.records = () if release else (
                NS(kind="service", start_ps=0, end_ps=0, cause_job_id="z", predecessor_id=None,
                   limiting_resource_id=0, category="overhead"),)
        elif fault == "dispatch":
            bad.observations.events.records = events[:-1]
        else:
            bad.simulator_metrics.event_count = 4
        with pytest.raises(ValueError):
            validate_fixed_delay_observations(bad, job)
        validate_fixed_delay_observations(r, job)  # restored structural positive, not native


def test_literal_same_time_release_dependency_inline_zero_partition() -> None:
    """Hand-entered partial subrecords, not an executed or serializable native result."""
    import copy
    from types import SimpleNamespace as NS
    from typing import Any

    from uarch_contract.mapped import (
        ClockSetup, FixedDelay, JobObservation, Observed, ResourceInterval,
        validate_fixed_delay_observations,
    )
    # id, owner, duration, release, ready, grant, end, tight causes
    literals = (
        ("a", 0, 1000, 0, 0, 0, 1000, ()),
        ("b", 0, 0, 1000, 1000, 1000, 1000, ("a",)),
        ("c", 0, 0, 0, 1000, 1000, 1000, ("a", "b")),
        ("d", 1, 1000, 0, 1000, 1000, 2000, ("b", "c")),
    )
    records = tuple(JobObservation(
        job_id=k, owner_id=o, release_ps=release, ready_ps=ready, start_ps=start, end_ps=end,
        blocking_predecessor_ids=causes, op_id=None, group_id=None, repeat_index=None,
        tile_origin=None, tile_extents=None, useful_matrix_ops=0, issued_matrix_ops=0,
        vector_arithmetic_ops=0, cast_ops=0,
    ) for k, o, _, release, ready, start, end, causes in literals)
    declared = tuple(FixedDelay(kind="fixed_delay", id=k, owner_id=o, duration_ps=duration)
                     for k, o, duration, *_ in literals)
    deps = tuple(NS(before=a, after=b) for a, b in (("a", "c"), ("b", "d"), ("c", "d")))
    job: Any = NS(clock_setup=(ClockSetup(job_id="b", domain="core", start_ps=1000, n_cycles=0),),
                  freq_hz=NS(core=10**9), mapped_point=NS(task_graph=NS(jobs=declared, dependencies=deps)))
    # Explicit actual-dispatch ordering premise. This test supplies no scheduler.
    dispatches = (
        (0, 1, "a", "TASK_READY"), (0, 2, "a", "OWNER_GRANT"),
        (1000, 0, "a", "COMPUTE_DONE"), (1000, 1, "b", "TASK_RELEASE"),
        (1000, 1, "c", "DEPENDENCY_READY"), (1000, 2, "b", "OWNER_GRANT"),
        (1000, 2, "b", "COMPUTE_DONE"), (1000, 2, "d", "DEPENDENCY_READY"),
        (1000, 2, "c", "OWNER_GRANT"), (1000, 2, "c", "COMPUTE_DONE"),
        (1000, 2, "d", "DEPENDENCY_READY"), (1000, 2, "d", "OWNER_GRANT"),
        (2000, 0, "d", "COMPUTE_DONE"),
    )
    owners = {r.job_id: r.owner_id for r in records}
    events = tuple(EventObservation.model_validate(dict(
        event_seq=i, observation_ordinal=i, t_ps=t, phase=p, target_id=owners[k], kind=kind, job_id=k,
    )) for i, (t, p, k, kind) in enumerate(dispatches))
    queues = tuple(sorted((QueueObservation(
        queue_id=f"owner:{r.owner_id}:{p}", item_id=r.job_id, population=cast(Literal["waiting", "service"], p), enter_ps=a, leave_ps=b,
        owner_id=r.owner_id, clock_domain=None,
    ) for r in records for p, a, b in (
        ("waiting", r.ready_ps, r.start_ps), ("service", r.start_ps, r.end_ps),
    )), key=lambda q: (q.queue_id, q.item_id, q.population)))
    intervals = tuple(ResourceInterval(owner_id=r.owner_id, job_id=r.job_id, start_ps=r.start_ps,
        end_ps=r.end_ps, activity="fixed_delay", limiting_resource_id=r.owner_id,
        service_start_ps_ratio=None, service_end_ps_ratio=None)
        for r in records if r.end_ps > r.start_ps)
    critical = (
        CriticalInterval(kind="service", start_ps=0, end_ps=1000, cause_job_id="a",
            predecessor_id=None, limiting_resource_id=0, category="overhead"),
        CriticalInterval(kind="service", start_ps=1000, end_ps=2000, cause_job_id="d",
            predecessor_id="b", limiting_resource_id=1, category="overhead"),
    )
    clock = ClockObservation(job_id="b", domain="core", grant_ps=1000, n_cycles=0,
        raw_edge_outcome="success", aligned_start_ps=1000, completion_ps=1000)
    r: Any = NS(duration_ps=2000, simulator_metrics=NS(event_count=13, max_live_events=2),
        observations=NS(dma=NS(status="unrepresented"), transfers=NS(status="unrepresented"),
            allocations=NS(status="unrepresented"), jobs=NS(records=records), events=NS(records=events),
            clocks=Observed[ClockObservation](status="observed", records=(clock,)),
            queue_membership=Observed[QueueObservation](status="observed", records=queues),
            resource_intervals=NS(records=intervals), critical_path=NS(records=critical)))
    validate_fixed_delay_observations(r, job)
    for fault in ("release_tie", "zero_bridge", "notification", "positive_phase", "fifo"):
        bad = copy.deepcopy(r)
        if fault == "release_tie":
            bad.observations.jobs.records = (
                records[0], records[1].model_copy(update={"blocking_predecessor_ids": ()}),
                records[2], records[3],
            )
        elif fault == "zero_bridge":
            bad.observations.critical_path.records = (
                critical[0], critical[1].model_copy(update={"predecessor_id": "a"}),
            )
        elif fault == "notification":
            bad.observations.events.records = events[:7] + events[8:]
        elif fault == "positive_phase":
            bad.observations.events.records = events[:-1] + (events[-1].model_copy(update={"phase": 2}),)
        else:
            # A phase-1-ready c cannot jump ahead of already enqueued b on owner0.
            changed = list(events)
            changed[5] = events[5].model_copy(update={"job_id": "c"})
            changed[6] = events[6].model_copy(update={"job_id": "c"})
            changed[8] = events[8].model_copy(update={"job_id": "b"})
            changed[9] = events[9].model_copy(update={"job_id": "b"})
            bad.observations.events.records = tuple(changed)
            bad.observations.jobs.records = (
                records[0], records[1].model_copy(update={"blocking_predecessor_ids": ("a", "c")}),
                records[2].model_copy(update={"blocking_predecessor_ids": ("a",)}), records[3],
            )
        with pytest.raises(ValueError):
            validate_fixed_delay_observations(bad, job)
        validate_fixed_delay_observations(r, job)
