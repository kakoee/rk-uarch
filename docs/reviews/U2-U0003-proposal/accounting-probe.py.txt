   m,s,q=r['model'],r['model_shape'],r['query'];d=m['d_model'];h=d//m['n_heads'];layers=m['n_layers'];a=m['experts_per_token'] or 1

   entry={'fixture_id':r['id'],'component_params_file':r['component_params_file'],'component_params_sha256':r['component_params_sha256'],'query':q,'precision':r['precision'],'tp':r['tp'],'frozen_counts':r['counts'],'frozen_duration_s':r['duration_s'],'frozen_duration_method':r['duration_method'],'frozen_omissions':r['omissions'],'matrix_ops_full_square_all_token_head':independent,'matrix_raw_rel':raw,'best_possible_residual_magnitude_with_5_percent_spend':min_residual,'count_budget_arithmetic_can_pass':min_residual<=.005,'linear_weight_elements':linear_weights,'attention_matrix_ops':attention_ops,'embedding_gather_matrix_ops':0,'embedding_table_elements':s['vocab_size']*d,'weight_gather_kv_read_lower_bound_bytes':read_lower_bound,'known_append_kv_write_bytes':kv_writes,'actual_producer_duration_s':None,'actual_engine_duration_s':None,'duration_parity':'not run'}
