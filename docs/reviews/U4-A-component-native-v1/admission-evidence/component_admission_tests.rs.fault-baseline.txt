//! PRIVATE SYNTHETIC admission only. Tests never dispatch or establish host authority.
use crate::{admission::Trust,component_admission::{self,TaskKind},hardware,wire};
use serde_json::{Value,json};
use std::collections::{BTreeMap,BTreeSet};
fn fixture(id:&str)->Value {let v:Value=serde_json::from_str(include_str!("../testdata/synthetic-component-jobs.json")).unwrap();v[id].clone()}
fn trust(v:&Value)->Trust {Trust{registry:v["registry"]["registry_hash"].as_str().unwrap().into(),profiles:BTreeMap::from([(v["profile"]["profile_hash"].as_str().unwrap().into(),BTreeSet::from([v["admission"]["case_id"].as_str().unwrap().into()]))]),engine:v["engine"].clone()}}
fn seal(v:&mut Value,key:&str){v.as_object_mut().unwrap().remove(key);v[key]=json!(hardware::digest(v).unwrap());}
fn reseal(v:&mut Value){
 for p in v["workload_closure"]["points"].as_array_mut().unwrap(){seal(&mut p["task_graph"],"graph_hash");seal(p,"payload_hash");}
 v["mapped_point"]=v["workload_closure"]["points"][0].clone();
 v["point_hash"]=v["mapped_point"]["payload_hash"].clone();
 v["profile"]["entries"][0]["graph_hash"]=v["mapped_point"]["task_graph"]["graph_hash"].clone();
 v["profile"]["entries"][0]["point_hash"]=v["point_hash"].clone();
 seal(&mut v["profile"],"profile_hash");
 v["admission"]["profile_hash"]=v["profile"]["profile_hash"].clone();
 v["workload_closure"]["admission"]=v["admission"].clone();
 seal(&mut v["workload_closure"],"bundle_hash");v["bundle_hash"]=v["workload_closure"]["bundle_hash"].clone();seal(v,"job_hash");
}
fn check(v:&Value)->crate::carrier::Result<component_admission::Admitted>{component_admission::validate(&wire::canonical(v).unwrap(),&trust(v))}
fn refusal(v:&Value,code:&str){assert_eq!(check(v).unwrap_err().code,code);}
