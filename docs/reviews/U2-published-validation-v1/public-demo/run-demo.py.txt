def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def inventory(root):
    return {str(p.relative_to(root)): {'sha256': digest(p), 'bytes': p.stat().st_size}
            for p in sorted(root.rglob('*')) if p.is_file()}

commands = []


def run(label, argv, blocked=False, expected_refusal=False):
    local = env.copy()
    if blocked:
        local['PYTHONPATH'] = f'{trap}:{S}/src:{S}/contract'
        prefix = [str(PY), '-c', 'import guard; from rkuarch.cli import app; app()']
    else:
        prefix = [str(PY), '-m', 'rkuarch.cli']
    argv = [str(a) for a in argv]
    start = time.monotonic()
    p = subprocess.run(prefix + argv, cwd=S, env=local, text=True,
                       stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    (E / (label + '.log')).write_text(p.stdout)
    row = {'label': label, 'argv': prefix + argv, 'cwd': str(S),
           'PYTHONPATH': local['PYTHONPATH'], 'returncode': p.returncode,
           'seconds': round(time.monotonic() - start, 3), 'producers_disabled': blocked,
           'expected_refusal': expected_refusal}
    commands.append(row)
    (E / 'commands.json').write_text(json.dumps(commands, indent=2) + '\n')
    print(label, p.returncode, flush=True)
    if blocked:
        assert 'U2_DEMO_GUARD_INSTALLED' in p.stdout, p.stdout
    if expected_refusal:
        assert p.returncode != 0 and 'U2_DEMO_PRODUCER_DISABLED' in p.stdout, p.stdout
    else:
        assert p.returncode == 0, p.stdout


def assemble(dest):
    run('assemble-' + dest.name, [
        'companions', 'assemble', '--prepared-input', D / 'prepared.json',
        '--assumptions', D / 'assumptions.json', '--capture-dir', D / 'capture',
        '--recipes', D / 'draft/metric-dependencies.json',
        '--recipe-review', R / 'external-review/recipe-review.json',
        '--registry', D / 'draft/registry.json',
        '--registry-review', R / 'external-review/registry-review.json',
        '--model-card', D / 'draft/model-card.json', '--no-comparisons',
        '--artifact-dir', dest / 'companions', '--output', dest / 'context.json',
    ], blocked=True)
