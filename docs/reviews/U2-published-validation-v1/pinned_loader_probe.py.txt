import json,os,subprocess,hashlib

env=dict(os.environ);env.pop('PYTHONPATH',None);env.pop('MYPYPATH',None);env['PYTHONDONTWRITEBYTECODE']='1'
args=[str(py),'-m','rkuarch.cli','rk-component','hw/designs/npu-l4.yaml','--component-id','compute.asic.npu-l4','--execution-model','contract/tests/fixtures/u2_b/a2-export/npu-l4.execution-model.json','--oracle-compat','--output',str(output)]

assert p.returncode==0

vendor=repo/'contract/vendor/rk-sim@1e5706e0ebfcc67c1a7333079a35b75f693e9963'
# Isolated child sees only the committed pinned loader, not any upstream checkout.
probe='''from pathlib import Path
