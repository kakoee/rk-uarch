def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value

def preflight():
    entries = subprocess.check_output(['git', 'ls-tree', '-r', '-z', BASE], cwd=INTEGRATION).split(b'\0')
    count = 0
    for entry in entries:
        if not entry:
            continue
        meta, name = entry.split(b'\t', 1)
        mode, kind, oid = meta.decode().split()
        assert kind == 'blob'
        path = ROOT / name.decode()
        data = path.read_bytes()
        actual = hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest()
        assert actual == oid, str(path)
        count += 1
    assert len(ui.support_files()) == 87
    assert h.sha(ui.ADOPTED / 'MANIFEST.json') == MANIFEST and h.sha(AUTHORITY) == REVIEW
    checked = refresh.audit_candidate(ui.ADOPTED)
    assert checked['input_manifest_sha256'] == INPUTS
    refresh.adoption_authority(ui.ADOPTED, AUTHORITY, REVIEW)
    return dict(audit=checked, validation_tree=BASE, clean_export_files=count, support_files=87, review_sha256=REVIEW, driver_sha256=h.sha(Path(__file__)), helpers_sha256=h.sha(HELPER), surface_checker_sha256=h.sha(SURFACE), host=platform.node(), platform=platform.platform(), python=sys.version, argv=sys.argv, environment={k: os.environ.get(k) for k in ('PYTHONPATH', 'PYTHONDONTWRITEBYTECODE', 'PYTHONHASHSEED', 'OMP_NUM_THREADS')})
