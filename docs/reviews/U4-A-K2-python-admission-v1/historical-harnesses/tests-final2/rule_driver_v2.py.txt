def deny(event, args):
    if event in {
        "subprocess.Popen",
        "os.system",
        "os.exec",
        "os.posix_spawn",
        "os.spawn",
        "os.fork",
        "os.forkpty",
    }:
        raise RuntimeError("Python diagnostic prohibits process execution: " + event)
