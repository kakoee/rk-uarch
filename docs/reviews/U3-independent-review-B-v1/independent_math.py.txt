from fractions import Fraction as F
from math import ceil


def rope(B, S, Hq, Hkv, D, R, L, act_bytes, casts, coef_rows, coef_bytes=4,
         per_head_coef=False, drop_tail=False, repeat_applied=1):
    M = B * S
    rot = {"q": M * Hq * R, "k": M * Hkv * R}
    ops_per_scalar = 3 + (2 if casts else 0)
    vec = sum(rot.values()) * ops_per_scalar * L * repeat_applied
    coef_per_node = M * 2 * (R // 2) * coef_bytes  # cos+sin, R/2 frequencies each
    heads = {"q": Hq, "k": Hkv}
    width = R if drop_tail else D
    reads = writes = 0
    for n in ("q", "k"):
        act = M * heads[n] * D * act_bytes
        reads += act + coef_per_node * (heads[n] if per_head_coef else 1)
        writes += M * heads[n] * width * act_bytes
    alloc = coef_rows * (R // 2) * 2 * coef_bytes
    return vec, reads * L, writes * L, alloc

# Clock edges at 1.2 GHz, exact rational ceil; build-spec next_edge(t, n) = t_{k+n}.
def t(k, f=1_200_000_000):
    return ceil(F(k * 10**12, f))


def next_edge(tp, n, f=1_200_000_000):
    k = 0
    while t(k, f) < tp:
        k += 1
    k_formula = 0 if tp == 0 else (tp - 1) * f // 10**12 + 1  # A's integer-only index
    assert k == k_formula
    return t(k + n, f)
