#!/usr/bin/env bash
# Human-only oracle generation. Agents must not execute this wrapper.
# Freeze preparation is authorized; candidate adoption/commit/push/closure remain separate.
set -euo pipefail
cd /home/jjaff/AI-infra-simulation/rk-uarch-u2-integration
U2_FREEZE="$PWD/docs/reviews/U2-source-input-freeze-v3"
U2_LOGS="$PWD/docs/reviews/U2-real-generation-v3"
U2_OUTPUT="$PWD/tables/U2-real-generation-v3"
U2_CANDIDATE="$U2_OUTPUT/candidate"
U2_PYTHON=/home/jjaff/AI-infra-simulation/rk-uarch/.venv/bin/python
export PATH="/home/jjaff/.local/bin:$PATH"
export PYTHONDONTWRITEBYTECODE=1
export PYTHONPATH="$PWD/src:$PWD/contract:$PWD"
export UV_PROJECT_ENVIRONMENT=/tmp/rk-sim-u1-jjaffari-1e5706e
unset PARAMS
"$U2_PYTHON" -B "$U2_FREEZE/preflight.py" --require-freeze-record
test ! -e "$U2_LOGS" && test ! -L "$U2_LOGS"
test ! -e "$U2_OUTPUT" && test ! -L "$U2_OUTPUT"
mkdir "$U2_LOGS"
mkdir "$U2_OUTPUT"
"$U2_PYTHON" -B "$U2_FREEZE/preflight.py" --require-freeze-record > "$U2_LOGS/preflight.json"
git rev-parse HEAD > "$U2_LOGS/local-HEAD.txt"
git branch --show-current > "$U2_LOGS/local-branch.txt"
U2_COMMAND=("$U2_PYTHON" -B -m scripts.vendor_rk
  --sha 1e5706e0ebfcc67c1a7333079a35b75f693e9963
  --rk /home/jjaff/AI-infra-simulation/rk-sim-u1-pin
  --inputs "$U2_FREEZE/inputs"
  --input-manifest-sha256 08d0e905cc50943d286b9c6dbcfb2cf844c969dda28b9446162d12734870871c
  --output-root "$U2_CANDIDATE")
UARCH_HUMAN=1 "${U2_COMMAND[@]}" 2>&1 | tee "$U2_LOGS/created.log"
"$U2_PYTHON" -B - "$U2_LOGS/created.log" "$U2_CANDIDATE" created <<'CHECK'
import sys
from pathlib import Path
assert f'{sys.argv[3]}: 48 files at {sys.argv[2]}' in Path(sys.argv[1]).read_text().splitlines()
CHECK
"$U2_PYTHON" -B "$U2_FREEZE/preflight.py" --require-freeze-record > "$U2_LOGS/preflight-between.json"
UARCH_HUMAN=1 "${U2_COMMAND[@]}" 2>&1 | tee "$U2_LOGS/verified-identical.log"
"$U2_PYTHON" -B - "$U2_LOGS/verified-identical.log" "$U2_CANDIDATE" verified-identical <<'CHECK'
import sys
from pathlib import Path
assert f'{sys.argv[3]}: 48 files at {sys.argv[2]}' in Path(sys.argv[1]).read_text().splitlines()
CHECK
"$U2_PYTHON" -B "$U2_FREEZE/preflight.py" --require-freeze-record > "$U2_LOGS/preflight-after.json"
"$U2_PYTHON" -B -m scripts.vendor_rk --check --output-root "$U2_CANDIDATE" > "$U2_LOGS/strict-check.txt" 2>&1
"$U2_PYTHON" -B -m tests.u2_refresh inspect "$U2_CANDIDATE" > "$U2_LOGS/raw-inspection.json"
"$U2_PYTHON" -B -m scripts.u2_inputs compare "contract/vendor/rk-sim@1e5706e0ebfcc67c1a7333079a35b75f693e9963" "$U2_CANDIDATE" > "$U2_LOGS/semantic-diff.json"
"$U2_PYTHON" -B - "$U2_CANDIDATE" "$U2_LOGS" <<'CHECK'
import hashlib, sys
from pathlib import Path
manifest=Path(sys.argv[1])/'MANIFEST.json'
line=f'{hashlib.sha256(manifest.read_bytes()).hexdigest()}  {manifest}\n'
with (Path(sys.argv[2])/'candidate-manifest-sha256.txt').open('x') as stream:
    stream.write(line)
print(line,end='')
CHECK
printf '%s\n' 'STOP: return both generation status lines and the candidate manifest digest for coordinator audit. Adoption remains separate.'
