import datetime, json, os, pathlib, shlex, subprocess, sys, time
ROOT = pathlib.Path(__file__).resolve().parent
CLONE = ROOT / 'checkout'
ENV = os.environ.copy()
ENV.pop('VIRTUAL_ENV', None)
ENV['UV_CACHE_DIR'] = str(ROOT / 'uv-cache')
ENV['UV_PYTHON_INSTALL_DIR'] = str(ROOT / 'uv-python')
ENV['UV_PROJECT_ENVIRONMENT'] = str(CLONE / '.venv')
def run(label, argv, cwd=None):
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    path = ROOT / 'logs' / (label + '.log')
    command = shlex.join(argv)
    print('\n$ ' + command, flush=True)
    with path.open('w') as log:
        log.write('UTC start: ' + started + '\nCWD: ' + str(cwd or ROOT) + '\n$ ' + command + '\n')
        log.flush()
        t = time.monotonic()
        try:
            p = subprocess.Popen(argv, cwd=cwd or ROOT, env=ENV, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
            for line in p.stdout:
                log.write(line)
                log.flush()
                print(line, end='', flush=True)
            code = p.wait()
        except OSError as e:
            log.write(str(e) + '\n')
            print(str(e), flush=True)
            code = 127
        elapsed = time.monotonic() - t
        log.write(f'\nExit code: {code}\nElapsed seconds: {elapsed:.3f}\n')
    with (ROOT / 'results.jsonl').open('a') as out:
        out.write(json.dumps(dict(label=label, command=command, cwd=str(cwd or ROOT), started=started, exit_code=code, elapsed_seconds=round(elapsed,3), log=str(path))) + '\n')
    print(f'Exit code: {code}; elapsed: {elapsed:.3f}s', flush=True)
    return code
