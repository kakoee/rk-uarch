from pathlib import Path
import re, subprocess, os, json, hashlib, difflib
out=Path('/home/jjaff/AI-infra-simulation/rk-uarch-u2-integration/docs/reviews/U2-final-documentation-refresh')
source=Path('/tmp/U2-doc-refresh-1_ofh_p8/source')
py='/tmp/U2-github-cold-7fee0a8-g2i3kbng/venv/bin/python'
proposed=(out/'proposed-how-it-works.md').read_text()
blocks=re.findall(r'```sh\n(.*?)```',proposed,re.S)
env=dict(os.environ,PYTHONPATH=f'{source}/src:{source}/contract',PYTHONDONTWRITEBYTECODE='1',PATH=str(Path(py).parent)+':'+os.environ['PATH'])
env.pop('MYPYPATH',None)
results=[]
for i,block in enumerate(blocks):
 if 'u2-doc-demo-' in block:
  expected=(out/'reproduce-demo.py.txt').read_text()
  assert block=="python -B - <<'PY'\n"+expected+"PY\n"
  results.append({'block':i+1,'status':'passed exact embedded script in prior bounded demonstration','log':'demo.log'})
  continue
 p=subprocess.run(['bash','-c',block],cwd=source,env=env,capture_output=True,text=True)
 (out/f'command-{i+1}.log').write_text(p.stdout+p.stderr)
 results.append({'block':i+1,'exit_code':p.returncode,'log':f'command-{i+1}.log'})
 assert p.returncode==0,(i,p.stdout,p.stderr)
links=[]
for target in re.findall(r'\]\(([^)]+)\)',proposed):
 if '://' in target:continue
 base=target.split('#')[0]
 p=(source/'docs'/base).resolve()
 assert p.is_relative_to(source),(target,'escapes export')
 assert p.exists(),target
 links.append(target)
original=(source/'docs/how-it-works.md').read_text()
patch=''.join(difflib.unified_diff(original.splitlines(True),proposed.splitlines(True),fromfile='a/docs/how-it-works.md',tofile='b/docs/how-it-works.md'))
(out/'how-it-works.patch').write_text(patch)
p=subprocess.run(['git','apply','--check',str(out/'how-it-works.patch')],cwd=source,capture_output=True,text=True)
assert p.returncode==0,p.stderr
# Validate reconstruction in memory without applying the patch anywhere.
import tempfile
with tempfile.TemporaryDirectory(prefix='u2-doc-patch-') as t:
 root=Path(t);(root/'docs').mkdir();(root/'docs/how-it-works.md').write_text(original)
 subprocess.run(['git','apply','--unsafe-paths',str(out/'how-it-works.patch')],cwd=root,check=True)
 assert (root/'docs/how-it-works.md').read_text()==proposed
result={'command_blocks':results,'relative_links_checked_against_published_export':len(links),'relative_links':links,'external_links':'CI URL supported by local receipts; not refetched','patch_check':'passed; temporary-document reconstruction equals proposal','proposal_sha256':hashlib.sha256(proposed.encode()).hexdigest(),'patch_sha256':hashlib.sha256(patch.encode()).hexdigest()}
(out/'document-checks.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='relative_links'},indent=2))
