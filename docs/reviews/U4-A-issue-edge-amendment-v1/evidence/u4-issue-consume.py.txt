import hashlib,json,traceback
from pathlib import Path
from uarch_contract.mapped import MappedEngineJob,validate_result
from validation.component.observations_v3 import check_v3
root=Path('/tmp/u4-issue-amendment-native')
expected=json.loads(Path('docs/reviews/U4-B-component-stage2-v1/first-use-bindings.json').read_text())['observation_contract']
records=[]
for path in sorted((root/'source/probes').glob('*.json')):
    result=root/'evidence/target'/path.name
    if not result.exists(): continue
    j=json.loads(path.read_bytes());r=json.loads(result.read_bytes());row={'case':path.stem,'input_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'output_sha256':hashlib.sha256(result.read_bytes()).hexdigest()}
    for name,fn in [('validator',lambda:validate_result(r,MappedEngineJob.model_validate(j))),('checker',lambda:check_v3(j,r,expected))]:
        try:fn();row[name]='pass'
        except Exception as e:row[name]=str(e);traceback.print_exc()
    records.append(row)
print(json.dumps(records,indent=2))
Path('/tmp/u4-issue-consumption.json').write_text(json.dumps(records,indent=2)+'\n')
raise SystemExit(any(x[k]!='pass' for x in records for k in ('validator','checker')))
